# DemonCAT v0.2 Fault Catalog Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement v0.2 — 16 new faults (network 10, process 4, cpu 1, storage 1) via cnf+script (no binary per-fault changes), plus two core framework enhancements (`optional_params` field + inject-only dispatch branch) required to express the new catalog.

**Architecture:** v0.1 framework (cli/registry/executor/safety/state/config/output/dispatch) is stable. v0.2 adds two core capabilities (optional_params, inject-only), then adds 16 `[fault.<uid>]` sections to `demoncat.conf.example` + 16 shell scripts under `config/scripts/<module>/`, each verified by table-driven `mock_executor` tests asserting the dispatched command string + env vars. No fault requires recompiling `dcat` — only core enhancements touch `src/core/`.

**Tech Stack:** C11 (gnu11 extensions for `usleep`/`fork`/`select`), CMake ≥ 3.10, pthread, vendored cJSON, `-Wall -Wextra -Werror`, CTest with `WORKING_DIRECTORY=${CMAKE_SOURCE_DIR}` so `config/demoncat.conf.example` + `config/scripts/*.sh` resolve.

## Global Constraints

- Language: C11, CMake ≥ 3.10, `-Wall -Wextra -Werror`, static link, pthread.
- Third-party: only cJSON (vendored at `third_party/cjson/`); no new deps.
- Tests run with `WORKING_DIRECTORY=${CMAKE_SOURCE_DIR}` (dcat.conf.example + scripts resolve).
- `dcat` is one-shot CLI; background auto-clean thread dies on exit — `state_lazy_clean()` runs at startup.
- Params go via env vars (`DCAT_OP`/`DCAT_UID`/`DCAT_PARAM_<KEY>`), never argv (no shell injection).
- `duration` is **optional** for all recoverable faults; absent → no auto-recovery (manual clean).
- `inject`-only faults (`supported_ops=inject`) create no state record, no `record_id`, no clean/query.
- Follow existing code style: `core/*.c` include `"core/foo.h"`, tests include `"core/foo.h"`, no comments unless asked.
- Commit one task at a time; conventional commit messages (`feat:`, `test:`, `docs:`).

```sh
#!/bin/sh
# DemonCAT example fault: CPU overload (background mode).
# Env: DCAT_OP, DCAT_UID, DCAT_PARAM_CORES, DCAT_PARAM_DURATION
# v0.2: duration optional. If unset, run until SIGTERM (manual clean).
set -e

cleanup() {
    trap - TERM INT
    kill 0 2>/dev/null
    exit 0
}
trap cleanup TERM INT

case "${DCAT_OP:-inject}" in
    inject)
        cores=${DCAT_PARAM_CORES:-1}
        i=0
        while [ "$i" -lt "$cores" ]; do
            yes >/dev/null 2>&1 &
            i=$((i + 1))
        done
        if [ -n "$DCAT_PARAM_DURATION" ]; then
            sleep "$DCAT_PARAM_DURATION"
            cleanup
        else
            while true; do sleep 3600; done
        fi
        ;;
    clean)
        cleanup
        ;;
esac
```

```c
static int is_inject_only(const fault_def_t *f) {
    /* supported_ops contains no "clean" -> inject-only (one-shot, no state) */
    const char *p = f->supported_ops;
    while (*p) {
        const char *c = strchr(p, ',');
        size_t len = c ? (size_t)(c - p) : strlen(p);
        if (len == 5 && !strncmp(p, "clean", 5)) return 0;
        if (!c) break;
        p = c + 1;
    }
    return 1;
}

result_t *dispatch_inject(const fault_def_t *f, const params_t *p) {
    result_t *pc = safety_precheck(f, "inject", p);
    if (pc->code != 0) return pc;
    result_free(pc);

    char cmd[256];
    executor_build_cmd(f, "inject", p, cmd, sizeof cmd);
    executor_set_env("inject", f->uid, p);

    cJSON *data = cJSON_CreateObject();

    if (is_inject_only(f)) {
        result_t *rr = executor_run(cmd, 30000);
        if (rr->code != 0) {
            cJSON_Delete(data);
            result_free(rr);
            return result_err("inject", f->uid, DCAT_E_RUN, "script failed");
        }
        if (cJSON_GetObjectItem(rr->json ? cJSON_Parse(rr->json) : NULL, "message")) {
            /* best-effort: copy message from executor result */
        }
        result_free(rr);
        return result_ok("inject", f->uid, data);
    }

    if (state_find(f->uid))
        return result_err("inject", f->uid, DCAT_E_SAFETY, "already active");

    int dur = param_int(p, "duration");
    if (dur < 0) dur = f->timeout;

    pid_t bgpid = 0;
    if (f->exec_mode == EXEC_BACKGROUND) {
        pid_t pid = executor_spawn(cmd);
        if (pid < 0) { cJSON_Delete(data); return result_err("inject", f->uid, DCAT_E_RUN, "spawn failed"); }
        bgpid = pid;
        cJSON_AddNumberToObject(data, "pid", (double)pid);
    } else {
        result_t *rr = executor_run(cmd, dur > 0 ? dur * 1000 : 30000);
        if (rr->code != 0) {
            cJSON_Delete(data);
            result_free(rr);
            return result_err("inject", f->uid, DCAT_E_RUN, "script failed");
        }
        result_free(rr);
    }
    int id = state_add(f->uid, bgpid, dur);
    cJSON_AddNumberToObject(data, "record_id", id);
    return result_ok("inject", f->uid, data);
}
```

```c
result_t *dispatch_clean(const fault_def_t *f, const params_t *p) {
    result_t *pc = safety_precheck(f, "clean", p);
    if (pc->code != 0) return pc;
    result_free(pc);

    const injection_record_t *rec = state_find(f->uid);
    if (!rec) return result_err("clean", f->uid, DCAT_E_RUN, "no active injection");
    if (rec->bg_pid > 0) {
        executor_kill(rec->bg_pid);
    } else {
        char cmd[256];
        executor_build_cmd(f, "clean", p, cmd, sizeof cmd);
        executor_set_env("clean", f->uid, p);
        result_t *r = executor_run(cmd, 30000);
        result_free(r);
    }
    int rid = rec->record_id;
    state_mark_inactive(rid);
    cJSON *data = cJSON_CreateObject();
    cJSON_AddNumberToObject(data, "record_id", rid);
    return result_ok("clean", f->uid, data);
}
```

```sh
#!/bin/sh
# rNET_port_occupy: hold a TCP/UDP port (background).
# Env: DCAT_OP, DCAT_PARAM_PORT, DCAT_PARAM_PROTOCOL (default tcp)
cleanup() { trap - TERM INT; kill 0 2>/dev/null; exit 0; }
trap cleanup TERM INT

case "${DCAT_OP:-inject}" in
    inject)
        port=${DCAT_PARAM_PORT:?missing port}
        proto=${DCAT_PARAM_PROTOCOL:-tcp}
        # use python3 if available (portable socket hold); fallback to nc
        if command -v python3 >/dev/null 2>&1; then
            python3 -c "
import socket, time
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM if '$proto'=='tcp' else socket.SOCK_DGRAM)
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
s.bind(('0.0.0.0', $port))
if '$proto'=='tcp': s.listen(1)
while True: time.sleep(3600)
" &
        else
            # fallback: socat/nc; if neither, error
            (command -v socat >/dev/null && exec socat - TCP-LISTEN:$port,reuseaddr,fork) \
                || (command -v nc >/dev/null && exec nc -l -p $port) \
                || { echo "no python3/socat/nc to hold port" >&2; exit 1; }
        fi
        wait
        ;;
    clean)
        cleanup
        ;;
esac
```

```sh
# rNET_link_flap: loop link down/up (background, count-based).
# Env: DCAT_OP, DCAT_PARAM_IFACE, DCAT_PARAM_CYCLE_SEC (default 2), DCAT_PARAM_COUNT (default 10)
```

`config/scripts/process/proc_exit.sh`:

```sh
#!/bin/sh
# rPROC_exit: kill -9 a process (inject-only, irreversible).
# Env: DCAT_OP, DCAT_PARAM_PID
case "${DCAT_OP:-inject}" in
    inject)
        pid=${DCAT_PARAM_PID:?missing pid}
        if ! kill -9 "$pid" 2>&1; then
            echo "kill -9 $pid failed (no such process or no permission?)" >&2; exit 1
        fi
        echo "killed pid $pid"
        ;;
    clean)
        # inject-only: no clean op. If invoked, no-op (dcat rejects at precheck anyway).
        echo "rPROC_exit is inject-only; nothing to clean" >&2
        exit 0
        ;;
esac
```

`config/scripts/process/proc_dstate.sh`:

```sh
#!/bin/sh
# rPROC_dstate: spawn N processes doing uninterruptible IO (best-effort D-state).
# Env: DCAT_OP, DCAT_PARAM_COUNT
# Note: true D-state needs a stalled device/kernel module. v0.2 uses best-effort
# large sync writes (fsync blocks in D state briefly). Background; trap SIGTERM.
cleanup() { trap - TERM INT; kill 0 2>/dev/null; exit 0; }
trap cleanup TERM INT

case "${DCAT_OP:-inject}" in
    inject)
        count=${DCAT_PARAM_COUNT:?missing count}
        tmpf=/tmp/dcat.dstate.$$
        i=0
        while [ "$i" -lt "$count" ]; do
            # large writes + fsync enter D state during sync
            ( while true; do dd if=/dev/zero of="$tmpf.$i" bs=1M count=100 conv=fdatasync 2>/dev/null; done ) &
            i=$((i + 1))
        done
        wait
        ;;
    clean)
        cleanup
        ;;
esac
```

`config/scripts/process/proc_hang.sh`:

```sh
#!/bin/sh
# rPROC_hang: SIGSTOP a process; clean sends SIGCONT.
# Env: DCAT_OP, DCAT_PARAM_PID
SIDECAR=/run/demoncat/rPROC_hang.pid
case "${DCAT_OP:-inject}" in
    inject)
        pid=${DCAT_PARAM_PID:?missing pid}
        if ! kill -STOP "$pid" 2>&1; then
            echo "kill -STOP $pid failed" >&2; exit 1
        fi
        mkdir -p /run/demoncat 2>/dev/null
        echo "$pid" > "$SIDECAR" 2>/dev/null
        echo "stopped pid $pid"
        ;;
    clean)
        pid=${DCAT_PARAM_PID:-$(cat "$SIDECAR" 2>/dev/null || echo "")}
        if [ -n "$pid" ]; then
            kill -CONT "$pid" 2>/dev/null
        fi
        rm -f "$SIDECAR" 2>/dev/null
        echo "continued pid $pid"
        ;;
esac
```

`config/scripts/process/proc_zstate.sh`:

```sh
#!/bin/sh
# rPROC_zstate: spawn N zombie children (fork+exit without wait). Background.
# Env: DCAT_OP, DCAT_PARAM_COUNT
# Each child exits immediately; parent (this script) never waits -> children stay Z.
cleanup() { trap - TERM INT; kill 0 2>/dev/null; exit 0; }
trap cleanup TERM INT

case "${DCAT_OP:-inject}" in
    inject)
        count=${DCAT_PARAM_COUNT:?missing count}
        i=0
        while [ "$i" -lt "$count" ]; do
            # child: exit immediately; parent doesn't wait -> zombie
            ( : ) &
            i=$((i + 1))
        done
        # parent stays alive holding the zombies; wait for SIGTERM
        while true; do sleep 3600; done
        ;;
    clean)
        cleanup
        ;;
esac
```

```sh
#!/bin/sh
# rCPU_core_offline: offline CPU cores via sysfs.
# Env: DCAT_OP, DCAT_PARAM_CORES (e.g. "0,2,4" or "0-3")
# cpu0 is usually not offlinable; script skips and warns.
SIDECAR=/run/demoncat/rCPU_core_offline.list
parse_cores() {
    # expand "0-3" to "0 1 2 3"; pass "0,2,4" through
    echo "$1" | tr ',' ' ' | sed 's/-/ /g' | tr -s ' ' '\n'
}

case "${DCAT_OP:-inject}" in
    inject)
        spec=${DCAT_PARAM_CORES:?missing cores}
        offlist=""
        for n in $(parse_cores "$spec"); do
            path="/sys/devices/system/cpu/cpu$n/online"
            [ -w "$path" ] || { echo "cpu$n not offlinable (skipping)" >&2; continue; }
            if ! echo 0 > "$path" 2>&1; then
                echo "offline cpu$n failed" >&2; continue
            fi
            offlist="$offlist $n"
        done
        mkdir -p /run/demoncat 2>/dev/null
        echo "$offlist" > "$SIDECAR" 2>/dev/null
        echo "offlined cores:$offlist"
        ;;
    clean)
        offlist=$(cat "$SIDECAR" 2>/dev/null || echo "")
        for n in $offlist; do
            path="/sys/devices/system/cpu/cpu$n/online"
            [ -w "$path" ] && echo 1 > "$path" 2>/dev/null
        done
        rm -f "$SIDECAR" 2>/dev/null
        echo "onlined cores:$offlist"
        ;;
esac
```

```sh
#!/bin/sh
# rDISK_write_overload: spawn N dd writers (background).
# Env: DCAT_OP, DCAT_PARAM_DEVICE (path or block dev), DCAT_PARAM_WORKERS (default 4)
cleanup() { trap - TERM INT; kill 0 2>/dev/null; rm -f /tmp/dcat.write.* 2>/dev/null; exit 0; }
trap cleanup TERM INT

case "${DCAT_OP:-inject}" in
    inject)
        dev=${DCAT_PARAM_DEVICE:?missing device}
        workers=${DCAT_PARAM_WORKERS:-4}
        # if device is a directory, write temp files there; else write to /tmp
        if [ -d "$dev" ]; then
            target="$dev/dcat.stress"
        else
            target="/tmp/dcat.write.$$"
        fi
        i=0
        while [ "$i" -lt "$workers" ]; do
            ( while true; do dd if=/dev/zero of="${target}.${i}" bs=1M count=200 conv=fdatasync 2>/dev/null; done ) &
            i=$((i + 1))
        done
        wait
        ;;
    clean)
        cleanup
        ;;
esac
```
